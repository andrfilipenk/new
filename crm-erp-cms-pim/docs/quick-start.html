Quick Start Guide

Build your first application in 15 minutes

Introduction

This tutorial will guide you through creating a simple blog application with posts and comments. You'll learn the fundamentals of routing, controllers, models, and views.

Prerequisites: Make sure you have completed the installation and have a working framework instance.

Step 1: Create a Migration

First, let's create a migration for our posts table:

<?php
// database/migrations/2024_01_10_000001_create_posts_table.php

use Core\Database\Migration;
use Core\Database\Schema;

return new class extends Migration
{
    public function up()
    {
        Schema::create('posts', function($table) {
            $table->id();
            $table->string('title');
            $table->string('slug')->unique();
            $table->text('content');
            $table->enum('status', ['draft', 'published'])->default('draft');
            $table->integer('user_id');
            $table->timestamps();
        });
    }

    public function down()
    {
        Schema::drop('posts');
    }
};

Run the migration:

php cli/migrate.php up

Step 2: Create a Model

Create a Post model in app/Models/Post.php:

<?php

namespace App\Models;

use Core\ORM\Model;

class Post extends Model
{
    protected $table = 'posts';
    
    protected $fillable = [
        'title',
        'slug',
        'content',
        'status',
        'user_id'
    ];

    // Relationship to User
    public function user()
    {
        return $this->belongsTo(User::class);
    }
}

Step 3: Create a Controller

Create a controller to handle post operations:

<?php

namespace App\Controllers;

use App\Models\Post;
use Core\Request;
use Core\Response;
use Core\Validation\Validator;

class PostController extends Controller
{
    // List all posts
    public function index()
    {
        $posts = Post::where('status', 'published')
            ->orderBy('created_at', 'DESC')
            ->get();

        return Response::view('posts/index', [
            'posts' => $posts
        ]);
    }

    // Show single post
    public function show($slug)
    {
        $post = Post::where('slug', $slug)
            ->where('status', 'published')
            ->firstOrFail();

        return Response::view('posts/show', [
            'post' => $post
        ]);
    }

    // Show create form
    public function create()
    {
        return Response::view('posts/create');
    }

    // Store new post
    public function store(Request $request)
    {
        $validator = Validator::make($request->all(), [
            'title' => 'required|max:255',
            'content' => 'required',
            'status' => 'required|in:draft,published'
        ]);

        if ($validator->fails()) {
            return Response::redirect('/posts/create')
                ->with('errors', $validator->errors());
        }

        $slug = $this->generateSlug($request->input('title'));

        $post = Post::create([
            'title' => $request->input('title'),
            'slug' => $slug,
            'content' => $request->input('content'),
            'status' => $request->input('status'),
            'user_id' => auth()->id()
        ]);

        return Response::redirect('/posts/' . $post->slug)
            ->with('success', 'Post created successfully!');
    }

    private function generateSlug($title)
    {
        $slug = strtolower(trim(preg_replace('/[^A-Za-z0-9-]+/', '-', $title)));
        
        // Ensure uniqueness
        $count = Post::where('slug', 'LIKE', $slug . '%')->count();
        if ($count > 0) {
            $slug .= '-' . ($count + 1);
        }
        
        return $slug;
    }
}

Step 4: Define Routes

Add routes in app/routes.php:

<?php

use Core\Router;

// Post routes
Router::get('/posts', 'App\\Controllers\\PostController@index');
Router::get('/posts/create', 'App\\Controllers\\PostController@create');
Router::post('/posts', 'App\\Controllers\\PostController@store');
Router::get('/posts/{slug}', 'App\\Controllers\\PostController@show');
Note: Routes are matched in order. Place specific routes (like /posts/create) before dynamic routes (/posts/{slug}).

Step 5: Create Views

List View (views/posts/index.php)

<!-- views/posts/index.php -->
<?php $this->layout('layouts/main') ?>

<?php $this->section('content') ?>
<div class="container">
    <div class="d-flex justify-content-between align-items-center mb-4">
        <h1>Blog Posts</h1>
        <a href="/posts/create" class="btn btn-primary">Create Post</a>
    </div>

    <?php foreach($posts as $post): ?>
        <div class="card mb-3">
            <div class="card-body">
                <h2 class="card-title">
                    <a href="/posts/<?= $post->slug ?>"><?= htmlspecialchars($post->title) ?></a>
                </h2>
                <p class="text-muted">
                    Published on <?= date('F j, Y', strtotime($post->created_at)) ?>
                </p>
                <p class="card-text">
                    <?= htmlspecialchars(substr($post->content, 0, 200)) ?>...
                </p>
                <a href="/posts/<?= $post->slug ?>" class="btn btn-sm btn-outline-primary">Read More</a>
            </div>
        </div>
    <?php endforeach; ?>

    <?php if(empty($posts)): ?>
        <div class="alert alert-info">No posts found.</div>
    <?php endif; ?>
</div>
<?php $this->endSection() ?>
